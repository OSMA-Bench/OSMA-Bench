import numpy as np
from plan_utils import get_empty_velocity_map, set_pixel_by_radius, cm2index
from perception_utils import parse_query_obj

# Query: faster when on the right side of the table and slower when on the left side of the table.
velocity_map = get_empty_velocity_map(task='navigation')
table = parse_query_obj('table')
center_x, center_y, center_z = table.position
# faster on right side so 1.5 when y > center_y, slower on left side so 0.5 when y < center_y
velocity_map[:, center_y:] = 1.5
velocity_map[:, :center_y] = 0.5
ret_val = velocity_map

# Query: slow down by a quarter.
velocity_map = get_empty_velocity_map(task='navigation')
velocity_map[:] = 0.75
ret_val = velocity_map

# Query: slow down by a half when you're near the mug and bowl.
# IMPORTANT: pass the object (not .position) to set_pixel_by_radius — the
# function dilates the actual mesh by radius_cm. Passing .position works
# but produces a centroid-disk which is wrong for extended/grouped fixtures.
velocity_map = get_empty_velocity_map(task='navigation')
mug = parse_query_obj('mug')
set_pixel_by_radius(velocity_map, mug, radius_cm=53, value=0.5)
bowl = parse_query_obj('bowl')
set_pixel_by_radius(velocity_map, bowl, radius_cm=42, value=0.5)
ret_val = velocity_map
