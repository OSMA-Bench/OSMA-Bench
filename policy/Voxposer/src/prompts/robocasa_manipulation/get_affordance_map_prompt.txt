# "grasp the mug"), ALWAYS use this exact one-line pattern so the affordance is
# placed at the object; do NOT write affordance_map[x, y, z] = 1 with bare x,y,z:
#     obj = parse_query_obj('<object name>')
#     set_voxel_by_radius(affordance_map, obj.position, radius_cm=3, value=1)
# Only use explicit affordance_map[x, y, z] = 1 when the query gives literal
# [x, y, z] numbers. Every snippet must define its own variables — never rely on
# variables from a previous snippet.

# Query: a point 25cm past the drawer handle in the push direction.
affordance_map = get_empty_affordance_map()
handle = parse_query_obj('drawer handle')
frame = parse_query_obj('drawer frame')
# closing direction points from the open panel toward its frame; push in the
# HORIZONTAL plane only and aim slightly BELOW the handle so the gripper keeps
# pressing the drawer face instead of riding up over the handle.
push_dir = np.asarray(frame.position, dtype=float) - np.asarray(handle.position, dtype=float)
push_dir[2] = 0.0
push_dir = push_dir / (np.linalg.norm(push_dir) + 1e-9)
target = np.asarray(handle.position, dtype=float) + push_dir * cm2index(25, 'x')
target[2] = np.asarray(handle.position, dtype=float)[2] - cm2index(3, 'z')
set_voxel_by_radius(affordance_map, target, radius_cm=3, value=1)
ret_val = affordance_map
