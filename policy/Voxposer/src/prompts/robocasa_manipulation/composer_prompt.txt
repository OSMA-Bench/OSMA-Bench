# Query: open the microwave door.
# a HINGED door does NOT open by just moving near the handle — you must GRASP the
# handle and PULL it open (swing it toward you), then release. Three stages:
movable = parse_query_obj('gripper')
# 1) grasp the handle: face it, close the gripper around it
affordance_map = get_affordance_map('a point at the center of the microwave door handle')
rotation_map = get_rotation_map('face the microwave door handle')
gripper_map = get_gripper_map('open everywhere except 1cm around the microwave door handle')
execute(movable, affordance_map=affordance_map, rotation_map=rotation_map, gripper_map=gripper_map)
# 2) pull the grasped handle out to swing the door open (keep holding)
affordance_map = get_affordance_map('a point 35cm in front of the microwave door handle in the pull direction')
execute(movable, affordance_map=affordance_map)
# 3) release the handle and retract clear of the swung-open door
affordance_map = get_affordance_map('a point 20cm in front of the microwave door handle in the pull direction')
gripper_map = get_gripper_map('open everywhere')
execute(movable, affordance_map=affordance_map, gripper_map=gripper_map)

# Query: press the start button on the coffee machine.
movable = parse_query_obj('gripper')
# stage OUTSIDE the button along its outward face first — approaching from
# above/behind jams the wrist against the machine body and never touches it.
affordance_map = get_affordance_map('a point 8cm in front of the coffee machine start button, away from the coffee machine')
execute(movable, affordance_map=affordance_map)
affordance_map = get_affordance_map('a point 2cm into the coffee machine start button')
execute(movable, affordance_map=affordance_map)
# retract clear of the button: success needs the gripper >=15cm from the
# button AFTER the press registers — sitting on the button fails the check.
affordance_map = get_affordance_map('a point 20cm in front of the coffee machine start button, away from the coffee machine')
execute(movable, affordance_map=affordance_map)
